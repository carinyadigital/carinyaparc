ingredients:
  - item: "2 tbsp olive oil"
  - item: "1 large brown onion, diced"
  - item: "2 cloves garlic, minced"
  - item: "500 g mixed root vegetables (carrot, parsnip, potato), diced"
  - item: "1 tsp fresh rosemary, chopped (or 1/2 tsp dried)"
  - item: "1 L vegetable or chicken stock"
  - item: "400 g canned diced tomatoes"
  - item: "Salt and black pepper, to taste"
  - item: "Crusty bread, to serve"
instructions:
  - step: "Warm olive oil in a heavy pot over medium heat. Add onion and cook until softened, about 5 minutes."
  - step: "Stir in garlic and rosemary; cook for 1 minute until fragrant."
  - step: "Add root vegetables and stir to coat. Cook for 5 minutes."
  - step: "Pour in stock and tomatoes. Season with salt and pepper. Bring to a gentle simmer."
  - step: "Partially cover and simmer for 40–45 minutes until vegetables are tender."
  - step: "Adjust seasoning and serve with crusty bread."
